 'packages/sdk': {'bun': './src/index.ts', 'types': './dist/index.d.ts', 'default': './dist/index.js'},
 'packages/cli': {'bun': './src/main.ts', 'types': './dist/main.d.ts', 'default': './dist/main.js'},
}
failed = False
for directory, want in expected.items():
    data = json.loads(Path(directory + '/package.json').read_text())
    actual = data.get('exports', {}).get('.') if isinstance(data.get('exports'), dict) else None
    print(f"{directory}: actual exports={actual}")
    print(f"{directory}: expected exports={want}")
    if actual != want:
        failed = True
if failed:
